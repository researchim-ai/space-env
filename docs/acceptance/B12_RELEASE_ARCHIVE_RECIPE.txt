def normalize(info):
 info.uid=info.gid=info.mtime=0;info.uname=info.gname='';info.mode=0o644

def write_archive(name,items):
 path=out/name
 with path.open('wb') as raw:
  with gzip.GzipFile(filename='',mode='wb',fileobj=raw,mtime=0,compresslevel=6) as stream:
   with tarfile.open(fileobj=stream,mode='w') as tar:
    for arc,p in sorted(items):
     assert p.is_file() and not p.is_symlink(),p

     files[arc]={'bytes':p.stat().st_size,'sha256':file_digest(p)}
     tar.add(p,arcname=arc,recursive=False,filter=normalize)

 assert path.stat().st_size<95_000_000,'archive exceeds repository artifact policy; split finer'

 directories=sorted((root/'parts').iterdir()) if (root/'parts').exists() else sorted(p for p in root.iterdir() if p.is_dir())
 for directory in directories:
  items=[(profile+'/'+p.relative_to(root).as_posix(),p) for p in directory.rglob('*') if p.is_file()]
